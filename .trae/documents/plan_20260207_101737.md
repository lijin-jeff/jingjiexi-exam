# 修改 startNewPractice 方法，按 chunkSize 分批加载题目

## 问题分析

从用户提供的返回数据来看，`getQuestionStructure` 接口返回了 57 个题目，但是最终只返回了 10 个题目。

从之前的分析来看：
1. 我已经修改了 `startNewPractice` 方法，将 `fetchQuestionList(0, 10)` 改为 `fetchQuestionList(0, this.simplifiedQuestionList.length)`
2. 但是，这样会导致一次性加载所有 57 个题目，而不是按 `chunkSize` 分批加载

## 解决方案

需要修改 `startNewPractice` 方法，按 `chunkSize` 分批加载题目，而不是一次性加载所有题目。

### 1. 修改 startNewPractice 方法

修改 `startNewPractice` 方法，按 `chunkSize` 分批加载题目：

```javascript
// 构建精简的题目列表（包含uid和题型）
const simplifiedQuestionList = structure.map((item, index) => ({
  uid: item.uid,
  exam_type: item.exam_type,
  // 保持与原有结构兼容的字段
  is_answered: this.checkIfAnswered(item.uid),
  user_answer: this.getUserAnswer(item.uid),
  is_correct: this.getQuestionCorrectness(item.uid)
}));

console.log('[startNewPractice] 精简题目列表构建完成:', simplifiedQuestionList.length, '道题');

// 存储精简题目列表用于答题卡显示
this.simplifiedQuestionList = simplifiedQuestionList;

// 按chunkSize分批加载题目，而不是一次性加载所有题目
const chunkSize = 10;
const totalChunks = Math.ceil(simplifiedQuestionList.length / chunkSize);

// 加载第一块题目
this.fetchQuestionList(0, chunkSize);

// 延迟加载剩余的题目
for (let i = 1; i < totalChunks; i++) {
  setTimeout(() => {
    this.fetchQuestionList(i, chunkSize);
  }, i * 500); // 每块延迟500ms
}
```

### 2. 修改 fetchQuestionList 方法

修改 `fetchQuestionList` 方法，确保按 `chunkSize` 分批加载题目：

```javascript
async fetchQuestionList(chunkIndex = 0, chunkSize = 10) {
  // 防止重复加载
  if (this.isLoading) {
    console.warn('正在加载题目，跳过重复调用')
    return
  }
  
  performanceTester.startTest('Fetch Question List', '测试题目列表加载性能')
  const trackingInfo = performanceMonitor.startDataLoad('question_list')
  
  try {
    // 设置加载状态
    this.isLoading = true
    if (chunkIndex === 0) {
      uni.showLoading({ title: '加载中...' })
      console.log('设置加载状态为 true')
    }
    
    performanceTester.measure('start_api_call')
    
    let apiMethod = ''
    // 使用统一的参数构建方法
    let params = {
      // ✅ 修改：如果存在 simplifiedQuestionList，按指定索引范围获取题目UID
      ...(this.simplifiedQuestionList && this.simplifiedQuestionList.length > 0 && 
        chunkIndex !== undefined && chunkSize !== undefined) 
        ? {
            // 计算要获取的题目索引范围
            question_start_index: chunkIndex * chunkSize,
            question_end_index: Math.min((chunkIndex + 1) * chunkSize, this.simplifiedQuestionList.length),
            // 获取对应范围内的题目UID列表
            question_uids: this.simplifiedQuestionList
              .slice(chunkIndex * chunkSize, Math.min((chunkIndex + 1) * chunkSize, this.simplifiedQuestionList.length))
              .map(item => item.uid)
          }
        : this.buildApiParams(chunkIndex, chunkSize),
      ratio_params: this.ratioParams
    }
    
    // ✅ 根据是否有 question_uids 参数决定使用哪个 API 方法
    if (params.question_uids && params.question_uids.length > 0) {
      apiMethod = 'apiQuestionOrderListByUids';
    } else {
      // 根据题目类型选择不同的API
      const questionType = this.questionType || this.questionParams.questionType
      switch (questionType) {
        // 模考试题
        case 'mock':
          apiMethod = 'apiMockExaminationList'
          break
        // 正式考试（在线考试）
        case 'examination':
          apiMethod = 'apiExaminationQuestionList'
          break
        default:
          // 其他的（收藏、错题、顺序、章节练习等）全部默认按顺序获取题目
          apiMethod = 'apiQuestionOrderList'
      }
    }

    const res = await this.$api[apiMethod](params)
    
    // ✅ 增加调试日志：查看 API 返回的题目
    if (res && res.code === 1 && res.data) {
      // 获取实际的题目数据（兼容多种数据结构）
      const actualQuestionList = res.data.list || res.data.data || res.data;
      const actualCount = Array.isArray(actualQuestionList) ? actualQuestionList.length : 0;
      console.log('[fetchQuestionList] 加载题目数量:', actualCount, '道题');
    }
    
    // 从 API 返回结果获取总题数，确保总是使用最新的总题数
    if (res && res.data && res.data.total) {
      if (this.totalQuestionCount !== res.data.total) {
        this.totalQuestionCount = res.data.total;
      } 
    }
    
    performanceTester.measure('end_api_call', { apiMethod, questionCount: res?.data?.list?.length || 0 })
    
    if (res && res.code === 1 && res.data) {
      performanceTester.measure('start_init_questions')
      // 确保传递给processQuestionList的是有效的数组
      let rawQuestionList = []
      if (Array.isArray(res.data.list)) {
        rawQuestionList = res.data.list
        console.log('[题目解析] 从 res.data.list 读取:', rawQuestionList.length, '道题')
      } else if (Array.isArray(res.data)) {
        rawQuestionList = res.data
        console.log('[题目解析] 从 res.data 读取:', rawQuestionList.length, '道题')
      } else {
        console.warn('API返回的数据不是数组:', res.data)
        rawQuestionList = []
      }
      
      // 使用initQuestionList方法处理原始题目数据
      const questionList = this.initQuestionList(rawQuestionList)
      performanceTester.measure('end_init_questions', { questionCount: questionList.length })

      // 只有当题目列表非空时才更新
      if (questionList.length > 0) {
        // 按照simplifiedQuestionList的顺序插入新题目
        if (this.simplifiedQuestionList && this.simplifiedQuestionList.length > 0) {
          // 步骤1：为每个新题目添加其在 simplifiedQuestionList 中的目标索引
          questionList.forEach(newQuestion => {
            const targetIndex = this.simplifiedQuestionList.findIndex(item => item.uid === newQuestion.uid);
            if (targetIndex !== -1) {
              newQuestion._targetIndex = targetIndex;
            }
          });

          // 步骤2：创建一个临时数组，用于存储合并后的题目
          const mergedQuestions = [];

          // 步骤3：重建完整的题目列表，按照 simplifiedQuestionList 的顺序
          for (let i = 0; i < this.simplifiedQuestionList.length; i++) {
            const simplifiedQuestion = this.simplifiedQuestionList[i];
            // 在新题目列表中查找对应的题目
            const newQuestion = questionList.find(q => q.uid === simplifiedQuestion.uid);
            if (newQuestion) {
              // 如果找到，添加到合并列表
              mergedQuestions.push(newQuestion);
            } else {
              // 如果没有找到，检查是否已经在 swiperList 中
              const existingQuestion = this.swiperList.find(q => q.uid === simplifiedQuestion.uid);
              if (existingQuestion) {
                // 如果已经存在，添加到合并列表
                mergedQuestions.push(existingQuestion);
              }
            }
          }

          // 步骤4：处理在 simplifiedQuestionList 中找不到的题目（追加到末尾）
          questionList.forEach(newQuestion => {
            if (!newQuestion._targetIndex && !mergedQuestions.find(q => q.uid === newQuestion.uid)) {
              mergedQuestions.push(newQuestion);
            }
          });

          // 更新 swiperList
          this.swiperList = mergedQuestions;
          console.log('[题目加载] 按照 simplifiedQuestionList 顺序合并完成，总题数:', this.swiperList.length);
        } else {
          // 如果没有 simplifiedQuestionList，回退到原来的追加逻辑
          this.swiperList = [...this.swiperList, ...questionList];
        }

        // 更新当前题目索引
        if (this.currentQuestionIndex === -1 && this.swiperList.length > 0) {
          this.currentQuestionIndex = 0;
        }

        // 更新已答题统计
        this.updateAnsweredStats();
      }
    } else {
      console.warn('API返回失败:', res.msg)
      uni.showToast({
        title: '提示',
        icon: 'none',
        duration: 2000,
        mask: true
      })
    }

    // 清除加载状态
    this.isLoading = false
    if (chunkIndex === 0) {
      uni.hideLoading()
    }

    // 记录性能数据
    performanceMonitor.endDataLoad(trackingInfo, {
      success: res && res.code === 1,
      itemCount: res?.data?.list?.length || 0,
      duration: performanceTester.getDuration('Fetch Question List')
    })

    return res
  } catch (error) {
    console.error('[fetchQuestionList] 加载题目失败:', error)
    // 清除加载状态
    this.isLoading = false
    uni.hideLoading()

    // 记录性能数据
    performanceMonitor.endDataLoad(trackingInfo, {
      success: false,
      error: error.message,
      duration: performanceTester.getDuration('Fetch Question List')
    })

    throw error
  }
}
```

## 预期效果

1. **按 chunkSize 分批加载题目**：
   - 将 57 个题目按 `chunkSize`（例如 10）分成多个题包
   - 每次调用 `fetchQuestionList` 时，只加载一个题包的题目信息（例如 10 个）

   - 仍然使用 `simplifiedQuestionList` 和 `fetchQuestionList` 方法
