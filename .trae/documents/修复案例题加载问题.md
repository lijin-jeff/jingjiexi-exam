## 问题描述

点击57题（案例题），实际跳转到56题。

## 根本原因

案例题和普通题可能分开存储，API 按块加载时只返回普通题，案例题成为占位符。

## 修复方案

### 修改 `loadQuestionAndJump` 方法（第3480-3509行）

当按块加载找不到题目时，尝试单独加载该题目：

```javascript
// 加载题目并跳转
async loadQuestionAndJump(index, targetQuestionUid) {
  try {
    uni.showLoading({ title: '加载题目中...' });
    
    // 计算需要加载的块
    const chunkSize = this.preloadConfig.chunkSize || 10;
    const chunkIndex = Math.floor(index / chunkSize);
    
    // 加载题目
    await this.fetchQuestionList(chunkIndex, chunkSize);
    
    // 再次查找目标题目
    let newIndex = this.swiperList.findIndex(item => item.uid === targetQuestionUid);
    
    // 如果还是找不到，尝试单独加载该题目
    if (newIndex === -1) {
      console.log('[loadQuestionAndJump] 块加载未找到，尝试单独加载:', targetQuestionUid);
      await this.fetchSingleQuestion(targetQuestionUid);
      
      // 再次查找
      newIndex = this.swiperList.findIndex(item => item.uid === targetQuestionUid);
    }
    
    uni.hideLoading();
    
    if (newIndex !== -1) {
      console.log('[loadQuestionAndJump] 加载成功，跳转到:', newIndex);
      this.swiperCurrentIndex = newIndex;
      this.showAnswerCard = false;
    } else {
      console.error('[loadQuestionAndJump] 加载后仍未找到题目:', targetQuestionUid);
      uni.showToast({ title: '题目加载失败', icon: 'none' });
    }
  } catch (err) {
    console.error('[loadQuestionAndJump] 加载题目失败:', err);
    uni.hideLoading();
    uni.showToast({ title: '题目加载失败', icon: 'none' });
  }
}
```

### 新增 `fetchSingleQuestion` 方法

```javascript
// 单独加载一个题目
async fetchSingleQuestion(questionUid) {
  try {
    console.log('[fetchSingleQuestion] 单独加载题目:', questionUid);
    
    const res = await this.$func.request({
      url: this.$api.apiQuestionDetail,
      data: { question_uid: questionUid }
    });
    
    if (res && res.code === 1 && res.data) {
      // 处理题目数据
      const question = this.initQuestionList([res.data])[0];
      
      if (question && question.uid) {
        // 添加到 swiperList
        const currentList = examStore.state.swiperList || [];
        
        // 找到正确的位置插入
        const targetIndex = this.simplifiedQuestionList.findIndex(item => item.uid === question.uid);
        
        if (targetIndex !== -1) {
          // 创建新列表
          const newList = [...currentList];
          // 如果该位置是占位符，替换
          if (newList[targetIndex] && newList[targetIndex].isPlaceholder) {
            newList[targetIndex] = question;
          } else {
            // 否则追加
            newList.push(question);
          }
          
          examStore.setSwiperList(newList);
          console.log('[fetchSingleQuestion] 加载成功:', question.uid);
        }
      }
    }
  } catch (error) {
    console.error('[fetchSingleQuestion] 加载失败:', error);
  }
}
```
